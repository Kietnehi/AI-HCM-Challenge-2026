def resolve(owner: str, slug: str, inner: str = "", require: str = None) -> str:
    """Tim thu muc dataset.

    require: ten file BAT BUOC phai co trong thu muc ket qua (vd 'BUILD_MANIFEST.json').
    Kaggle doi khi giai nen dataset kem 1 lop thu muc con long them (vd
    'aic26-index/index/...'), nen chi os.path.exists() tren thu muc goc la KHONG du:
    no luon ton tai (la 1 folder) du file can nam sau hon 1-2 cap. Neu co ,
    tu do xuong (glob recursive) de tim dung thu muc chua no.
    """
    candidates = [f"{KG}/datasets/{owner}/{slug}", f"{KG}/{slug}"]
    candidates += glob.glob(f"{KG}/**/{slug}", recursive=True)
    seen = set()
    for base in candidates:
        if base in seen or not os.path.isdir(base):
            continue
        seen.add(base)
        if not require:
            p = os.path.join(base, inner) if inner else base
            if os.path.exists(p):
                return p
            continue
        hits = sorted(glob.glob(os.path.join(base, "**", require), recursive=True))
        if hits:
            found_dir = os.path.dirname(hits[0])
            return os.path.join(found_dir, inner) if inner else found_dir
    raise FileNotFoundError(
        f"Khong tim thay {owner}/{slug}" + (f" (chua ''{require}'')" if require else "") +
        f". Da attach dataset chua?
"
        f"Co san trong /kaggle/input: {os.listdir(KG) if os.path.isdir(KG) else 'KHONG CO'}"
    )
